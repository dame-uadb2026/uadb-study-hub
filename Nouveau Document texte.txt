# Définition du chemin de la base de données
BASE_DIR = os.path.dirname(os.path.abspath(__file__))
DB_PATH = os.path.join(BASE_DIR, 'uadb.db')

# 1. Initialisation de la base de données (exécutée automatiquement)
def init_db():
    conn = sqlite3.connect(DB_PATH)
    c = conn.cursor()
    
    # Table des matières
    c.execute('''
        CREATE TABLE IF NOT EXISTS matieres (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            nom TEXT NOT NULL,
            niveau TEXT NOT NULL,
            filiere TEXT NOT NULL,
            UNIQUE(nom, niveau, filiere)
        )
    ''')
    
    # Insertions par défaut si la table est vide
    c.execute('SELECT COUNT(*) FROM matieres')
    if c.fetchone()[0] == 0:
        matieres_initiales = [
            ("Physique Générale", "L1", "PC"),
            ("Chimie Organique", "L1", "PC"),
            ("Algèbre 1", "L1", "PC"),
            ("Analyse 1", "L1", "PC"),
            ("Thermodynamique", "L2", "PC"),
            ("Électromagnétisme", "L2", "PC"),
            ("Algorithmique & Programmation", "L1", "SID"),
            ("Introduction aux bases de données", "L1", "SID"),
            ("Statistiques Descriptives", "L1", "SID"),
            ("Bases de données avancées", "L2", "SID"),
            ("Développement Web", "L2", "SID")
        ]
        for nom, niveau, filiere in matieres_initiales:
            try:
                c.execute('INSERT INTO matieres (nom, niveau, filiere) VALUES (?, ?, ?)', (nom, niveau, filiere))
            except sqlite3.IntegrityError:
                pass
                
    conn.commit()
    conn.close()

# Exécution de la configuration initiale de la BDD
init_db()

# 3. Routes de l'application
@app.route('/')
def home():
    conn = sqlite3.connect(DB_PATH)
    c = conn.cursor()
    c.execute('SELECT id, nom, niveau, filiere FROM matieres ORDER BY filiere, niveau')
    matieres = c.fetchall()
    conn.close()
    return render_template_string(INDEX_HTML, matieres=matieres)
